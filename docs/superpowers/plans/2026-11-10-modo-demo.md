# Modo demo Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Un botón "Probar demo (sin registro)" que crea una cuenta temporal de 2 h con la rutina de Luciano, 6 semanas de historial y entreno hoy.

**Architecture:** Backend Laravel: columna `users.demo_expires_at`, middleware `demo.active`, servicio `CreateDemoAccount` (purga + tope + creación) y generador `DemoHistory`; ruta `POST /auth/demo`. Frontend Angular: `AuthService.startDemo()`, botón en el login, `DemoBanner` en el shell y menú de cuenta sin invitaciones.

**Tech Stack:** Laravel 13 + Pest (SQLite en memoria), Angular 22 + Vitest.

Spec: `docs/superpowers/specs/2026-11-10-modo-demo-design.md`.

## Global Constraints

- Duración: `CreateDemoAccount::LIFETIME_MINUTES = 120`. Tope: `MAX_ACTIVE = 50`. Límite: 5 demos por hora **por IP** (limitador con nombre `demo`; `throttle:5,60` contaría por usuario, y cada demo es un usuario nuevo).
- Fechas locales con `config('app.user_timezone')` (`America/Santiago`); BD en UTC.
- Textos de UI en español (Chile); código en inglés.
- Tests backend: `cd backend && ./vendor/bin/pest`. Frontend: `cd frontend && npx ng test --watch=false`.

```php
<?php

namespace App\Http\Middleware;

use Closure;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;
use Symfony\Component\HttpFoundation\Response;

/** Una demo vencida pierde la sesión: 401 como cualquier sesión cerrada. */
class EnsureDemoActive
{
    public function handle(Request $request, Closure $next): Response
    {
        $user = $request->user();
        if ($user?->isDemo() && $user->demo_expires_at->isPast()) {
            Auth::guard('web')->logout();
            if ($request->hasSession()) {
                $request->session()->invalidate();
            }

            return response()->json(['message' => 'La demo terminó.'], 401);
        }

        return $next($request);
    }
}
```

`DemoHistory.php`:

```php
<?php

namespace App\Services\Demo;

use App\Enums\SetType;
use App\Models\BodyMetric;
use App\Models\Routine;
use App\Models\RoutineDay;
use App\Models\RoutineDayExercise;
use App\Models\User;
use App\Models\Workout;
use App\Models\WorkoutExercise;
use Carbon\CarbonImmutable;
use Random\Engine\Mt19937;
use Random\Randomizer;

/** Historial ficticio pero creíble para la demo: 6 semanas de la rutina con progresión y peso corporal. */
class DemoHistory
{
    public const WEEKS = 6;

    private const MAX_SKIPPED = 2;

    /** Carga (kg) de las efectivas hace 6 semanas; 0 = solo peso corporal. */
    private const BASE_KG = [
        'Press de banca' => 70.0, 'Press con mancuernas' => 26.0, 'Press inclinado con mancuernas' => 22.0,
        'Elevaciones laterales' => 8.0, 'Dominadas' => 0.0, 'Remo con mancuerna' => 30.0,
        'Jalón al pecho' => 55.0, 'Curl de bíceps con mancuernas' => 12.0, 'Fondos' => 0.0,
        'Press de hombro con mancuernas' => 20.0, 'Extensión de tríceps en polea' => 25.0,
        'Sentadilla' => 80.0, 'Peso muerto rumano' => 70.0, 'Curl femoral' => 40.0, 'Gemelos en máquina' => 60.0,
    ];

    private const BILBO_KG = 40.0;

    private Randomizer $random;

    /** @var array<int, int> sesiones previas por ejercicio de la rutina */
    private array $sessions = [];

    public function generate(User $user, Routine $routine, CarbonImmutable $today): void
    {
        $this->random = new Randomizer(new Mt19937(2026));
        $this->sessions = [];
        $days = $routine->days()->with('exercises.exercise')->get()->keyBy('weekday');

        $skipped = 0;
        for ($ago = self::WEEKS * 7; $ago >= 1; $ago--) {
            $date = $today->subDays($ago);
            $day = $days->get($date->dayOfWeekIso);
            if ($day === null) {
                continue;
            }
            if ($skipped < self::MAX_SKIPPED && $this->random->getInt(1, 10) === 1) {
                $skipped++;
                continue;
            }
            $this->workout($user, $day, $date);
        }

        for ($week = self::WEEKS; $week >= 0; $week--) {
            BodyMetric::create([
                'user_id' => $user->id,
                'date' => $today->subWeeks($week)->toDateString(),
                'bodyweight_kg' => round(79.2 - 0.25 * (self::WEEKS - $week) + $this->random->getInt(-3, 3) / 10, 1),
            ]);
        }
    }

    private function workout(User $user, RoutineDay $day, CarbonImmutable $date): void
    {
        $start = $date->setTime(19, $this->random->getInt(0, 40))->utc();
        $workout = Workout::create([
            'user_id' => $user->id,
            'routine_day_id' => $day->id,
            'date' => $date->toDateString(),
            'started_at' => $start,
            'finished_at' => $start->addMinutes($this->random->getInt(55, 80)),
        ]);

        foreach ($day->exercises as $item) {
            $entry = WorkoutExercise::create([
                'workout_id' => $workout->id,
                'exercise_id' => $item->exercise_id,
                'position' => $item->position,
                'target_sets' => $item->target_sets,
                'target_reps_min' => $item->target_reps_min,
                'target_reps_max' => $item->target_reps_max,
                'target_text' => $item->target_text,
                'default_set_type' => $item->default_set_type,
                'rest_seconds' => $item->rest_seconds,
            ]);
            $session = $this->sessions[$item->id] = ($this->sessions[$item->id] ?? -1) + 1;
            $this->sets($entry, $item, $session);
        }
    }

    private function sets(WorkoutExercise $entry, RoutineDayExercise $item, int $session): void
    {
        $type = $item->default_set_type;
        $base = self::BASE_KG[$item->exercise->name] ?? 20.0;
        // Sube 2,5 kg cada 3 sesiones del ejercicio.
        $weight = $base > 0 ? $base + 2.5 * intdiv($session, 3) : 0.0;
        $number = 1;

        if ($type === SetType::Heavy) {
            foreach ([[0.5, 8], [0.7, 4]] as [$share, $reps]) {
                $this->set($entry, $number++, SetType::Warmup, round($weight * $share / 2.5) * 2.5, $reps);
            }
        }

        for ($i = 0; $i < ($item->target_sets ?? 3); $i++) {
            if ($type === SetType::Bilbo) {
                $this->set($entry, $number++, $type, self::BILBO_KG, 28 + $session + $this->random->getInt(0, 1));
                continue;
            }
            $reps = $this->random->getInt($item->target_reps_min ?? 8, $item->target_reps_max ?? 12);
            $this->set($entry, $number++, $type, $weight, $reps);
        }
    }

    private function set(WorkoutExercise $entry, int $number, SetType $type, float $weight, int $reps): void
    {
        $entry->sets()->create([
            'set_number' => $number,
            'set_type' => $type,
            'weight_kg' => $weight,
            'reps' => $reps,
        ]);
    }
}
```

```php
<?php

namespace App\Services\Demo;

use App\Models\Routine;
use App\Models\User;
use App\Models\Workout;
use Carbon\CarbonImmutable;
use Database\Seeders\GymTrackerSeeder;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Str;

/** Cuentas de "Probar demo": 2 h, con la rutina de ejemplo e historial; las vencidas se borran aquí. */
class CreateDemoAccount
{
    public const LIFETIME_MINUTES = 120;

    public const MAX_ACTIVE = 50;

    public function __construct(
        private readonly GymTrackerSeeder $seeder,
        private readonly DemoHistory $history,
    ) {}

    /** Sin tareas programadas (despertarían Aurora): se limpia al crear la siguiente demo. */
    public function purgeExpired(): int
    {
        $expired = User::query()->whereNotNull('demo_expires_at')->where('demo_expires_at', '<=', now())->pluck('id');
        foreach ($expired as $id) {
            // workout_exercises y routine_day_exercises restringen el borrado de exercises: primero esos.
            Workout::withoutGlobalScopes()->where('user_id', $id)->delete();
            Routine::withoutGlobalScopes()->where('user_id', $id)->delete();
            User::whereKey($id)->delete();
        }

        return $expired->count();
    }

    public function isFull(): bool
    {
        return User::query()->where('demo_expires_at', '>', now())->count() >= self::MAX_ACTIVE;
    }

    public function create(): User
    {
        return DB::transaction(function () {
            $user = User::create([
                'name' => 'Invitado',
                'email' => 'demo-'.Str::lower((string) Str::ulid()).'@demo.invalid',
                'password' => Str::random(40),
                'demo_expires_at' => now()->addMinutes(self::LIFETIME_MINUTES),
            ]);
            $today = CarbonImmutable::now(config('app.user_timezone'))->startOfDay();
            $this->seeder->run($user, $today->dayOfWeekIso);
            $this->history->generate($user, $user->routines()->firstOrFail(), $today);

            return $user;
        });
    }
}
```

```ts
import { ChangeDetectionStrategy, Component, DestroyRef, computed, effect, inject, signal } from '@angular/core';
import { Router } from '@angular/router';
import { AuthService } from '../../core/auth/auth.service';

/** "1 h 52 min" / "8 min". */
function formatLeft(ms: number): string {
  const minutes = Math.max(1, Math.ceil(ms / 60_000));
  const h = Math.floor(minutes / 60);
  const m = minutes % 60;
  return h ? `${h} h ${m} min` : `${m} min`;
}

/** Franja de la cuenta de demo: cuánto le queda y "Salir"; al vencer cierra la sesión. */
@Component({
  selector: 'app-demo-banner',
  changeDetection: ChangeDetectionStrategy.OnPush,
  template: `
    @if (expiresAt() !== null) {
      <div class="mb-4 flex items-center gap-3 rounded-full border border-primary/40 bg-primary/10 py-1.5 pr-1.5 pl-4 text-sm" role="status">
        <span class="flex-1"><b class="font-semibold text-primary">Modo demo</b> · se borra en {{ left() }}</span>
        <button type="button" class="h-9 rounded-full px-3 text-sm font-semibold text-primary underline" (click)="exit()">Salir</button>
      </div>
    }
  `,
})
export class DemoBanner {
  private readonly auth = inject(AuthService);
  private readonly router = inject(Router);

  private readonly now = signal(Date.now());
  protected readonly expiresAt = computed(() => {
    const iso = this.auth.user()?.demo_expires_at;
    return iso ? new Date(iso).getTime() : null;
  });
  protected readonly left = computed(() => formatLeft((this.expiresAt() ?? 0) - this.now()));

  constructor() {
    const handle = setInterval(() => this.now.set(Date.now()), 30_000);
    inject(DestroyRef).onDestroy(() => clearInterval(handle));
    effect(() => {
      const end = this.expiresAt();
      if (end !== null && this.now() >= end) void this.exit();
    });
  }

  protected async exit(): Promise<void> {
    try {
      await this.auth.logout();
    } catch {
      // Ya vencida: el servidor respondió 401; igual se vuelve al login.
    }
    await this.router.navigateByUrl('/login');
  }
}
```
